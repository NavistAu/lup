# lup fish completion
complete -c lup -s a -l all -d 'Print every hit, closest-first'
complete -c lup -s e -l echo -d 'Print file contents instead of paths'
complete -c lup -s g -l git -d 'Stop at git root'
complete -c lup -s r -l root -d 'Walk all the way to filesystem root'
complete -c lup -s f -l files -d 'Only match regular files'
complete -c lup -s d -l dirs -d 'Only match directories'
complete -c lup -s L -l follow -d 'Follow symlinks'
complete -c lup -s P -l no-follow -d 'Do not follow symlinks'
complete -c lup -s 0 -l null -d 'NUL-terminate each record'
complete -c lup -l relative -d 'Print paths relative to PWD'
complete -c lup -l completions -xa 'bash zsh fish' -d 'Print completion script'
complete -c lup -s h -l help -d 'Print help'
complete -c lup -s V -l version -d 'Print version'
